** Fix stale =LIBRARY_PATH= from the emacs-plus bundle

Rebuilds =LIBRARY_PATH= from Homebrew's version-independent
=gcc/current= path. emacs-plus hard-codes a versioned gcc path at build
time, which no longer exists after a gcc upgrade and makes =ld= warn on
every compile.

#+begin_src emacs-lisp
(when *is-a-mac*
  (let* ((versioned (car (last (file-expand-wildcards
                                "/opt/homebrew/lib/gcc/current/gcc/*/[0-9]*"))))
         (dirs (seq-filter #'file-directory-p
                           (delq nil (list versioned
                                           "/opt/homebrew/lib/gcc/current"
                                           "/opt/homebrew/lib")))))
    (when dirs
      (setenv "LIBRARY_PATH" (string-join dirs ":")))))
#+end_src

** Let project.el use projectile's roots

Registers projectile projects with the built-in =project.el=, so tools
built on =project-root= (such as the claude-code-ide MCP =project-info=)

** Remote (TRAMP) debugging

TRAMP logs at its default level 3. =my/tramp-debug-toggle= switches to
level 6, which logs all traffic to and from the remote end and mirrors
each =*debug tramp/...*= buffer to a file under
=temporary-file-directory=, so the log survives a force quit. Level 6
adds latency to every TRAMP call. Toggling closes cached connections so
the next one is logged from the handshake.

#+begin_src emacs-lisp
(setq tramp-verbose 3
      tramp-debug-to-file nil)

(defun my/tramp-debug-toggle ()
  "Toggle verbose TRAMP logging to debug buffers and to disk.
Closes cached connections so the next one is logged from the handshake up.
Open remote buffers reconnect on their next access."
  (interactive)
  (require 'tramp)
  (setq tramp-verbose (if (> tramp-verbose 3) 3 6)
        tramp-debug-to-file (> tramp-verbose 3))
  (tramp-cleanup-all-connections)
  (message "TRAMP verbose %d, file logging %s (%s)"
           tramp-verbose
           (if tramp-debug-to-file "on" "off")
           temporary-file-directory))
#+end_src

*** ESP-IDF / PlatformIO clangd

Inside ESP-IDF (=sdkconfig=) or PlatformIO (=platformio.ini=) projects,
uses the newest installed Espressif clangd, which supports the Xtensa
and RISC-V targets, and lets it query the PlatformIO cross-compilers.
Other C/C++ projects, or ESP projects without an Espressif clangd, use
the clangd on =PATH=.

#+begin_src emacs-lisp
(defun my/esp-project-p ()
  "Return non-nil if the current buffer is inside an ESP-IDF or PlatformIO project."
  (let ((dir (or buffer-file-name default-directory)))
    (or (locate-dominating-file dir "sdkconfig")
        (locate-dominating-file dir "platformio.ini"))))

(defun my/esp-clangd ()
  "Return the newest installed Espressif clangd, or nil if there is none."
  (car (last (sort (file-expand-wildcards
                    "~/.espressif/tools/esp-clang/*/esp-clang/bin/clangd" t)
                   #'string-version-lessp))))

(defun my/c-mode-setup ()
  "Point clangd at Espressif's build and toolchains in ESP projects."
  (when-let* (((my/esp-project-p))
              (clangd (my/esp-clangd)))
    ;; Defines the clangd options; this hook runs before lsp-mode loads.
    (require 'lsp-clangd)
    (setq-local lsp-clients-clangd-executable clangd)
    (setq-local lsp-clients-clangd-args
                (append lsp-clients-clangd-args
                        (list (concat "--query-driver="
                                      (expand-file-name
                                       "~/.platformio/packages/**/bin/*-elf-*")))))))

(add-hook 'c-mode-common-hook #'my/c-mode-setup)
#+end_src

*** Reload PlatformIO compile_commands.json for clangd

=my/pio-reload-compile-commands= regenerates =compile_commands.json=
with =pio run -t compiledb= and restarts the LSP workspace, so clangd
sees newly added =lib/= folders.

#+begin_src emacs-lisp
(defun my/pio-reload-compile-commands ()
  "Regenerate compile_commands.json via PlatformIO, then restart the LSP workspace.
Runs pio in a hidden buffer that is shown only if the build fails."
  (interactive)
  (let* ((default-directory (my/project-root))
         ;; Captured here, since the sentinel runs in the process buffer,
         ;; which has no LSP workspace.
         (workspaces (and (fboundp 'lsp-workspaces) (lsp-workspaces)))
         ;; Leading space keeps the buffer out of buffer lists.
         (buf (get-buffer-create " *pio-compiledb*")))
    (when (process-live-p (get-buffer-process buf))
      (user-error "pio compiledb is already running"))
    (with-current-buffer buf (erase-buffer))
    (message "Running pio compiledb...")
    (make-process
     :name "pio-compiledb"
     :buffer buf
     :command '("pio" "run" "-t" "compiledb" "-e" "unified")
     :sentinel
     (lambda (process _event)
       (when (memq (process-status process) '(exit signal))
         (if (zerop (process-exit-status process))
             (progn
               (message "pio compiledb done, restarting LSP")
               (mapc #'lsp-workspace-restart workspaces))
           (message "pio compiledb failed")
           (display-buffer (process-buffer process))))))))
#+end_src

** C#

*** Brace auto-formatting

Typing ={= opens an indented block on a new line, and =}= closes it,
leaving point on the empty line between the braces. Inside strings, and
after a space for ={= or after code for =}=, the brace is inserted as is.

#+begin_src emacs-lisp
(defun my/in-string-p ()
  "Return non-nil if point is inside a string."
  (nth 3 (syntax-ppss)))

(defun my/csharp-open-brace ()
  "Insert {, opening an indented block unless in a string or after a space."
  (interactive)
  (if (or (my/in-string-p)
          (eq (char-before) ?\s))
      (insert "{")
    (newline)
    (insert "{")
    (indent-according-to-mode)
    (newline)
    (indent-according-to-mode)))

(defun my/csharp-close-brace ()
  "Insert }, closing an indented block unless in a string or after code."
  (interactive)
  (if (or (my/in-string-p)
          (and (not (memq (char-before) '(?\s ?\t)))
               (save-excursion
                 (search-backward "{" nil t))))
      (insert "}")
    (newline)
    (insert "}")
    (indent-according-to-mode)
    (forward-line -1)
    (indent-according-to-mode)))

(add-hook 'csharp-mode-hook
          (lambda ()
            (local-set-key (kbd "{") #'my/csharp-open-brace)
            (local-set-key (kbd "}") #'my/csharp-close-brace)))
#+end_src

*** Live preview via impatient-mode + pandoc

=C-c C-c l= (=my/markdown-preview-toggle=, in place of markdown-mode's
eww-based live preview) serves the buffer over a local HTTP server,
rendered to HTML by pandoc with MathJax, and updates the page on every
change. The preview opens in an =xwidget-webkit= side window, or in the
external browser on builds without xwidget support. Toggling it off
stops serving the buffer, closes the xwidget window, and stops the
server once no buffer is being served.

#+begin_src emacs-lisp
(use-package simple-httpd)
(use-package impatient-mode)

(defun my/browse-url-in-emacs (url)
  "Open URL in an xwidget-webkit buffer, or externally without xwidget support."
  (if (featurep 'xwidget-internal)
      (xwidget-webkit-browse-url url)
    (browse-url url)))

(defun my/imp-markdown-filter (buffer)
  "Print BUFFER's markdown rendered to standalone HTML by pandoc."
  (princ (with-temp-buffer
           (insert-buffer-substring-no-properties buffer)
           (call-process-region (point-min) (point-max) "pandoc" t t nil
                                "-f" "markdown+tex_math_dollars"
                                "-t" "html5"
                                "--standalone"
                                "--mathjax")
           (buffer-string))))

(defun my/imp-live-url ()
  "URL that impatient-mode serves the current buffer at."
  (format "http://localhost:%d/imp/live/%s/"
          httpd-port (url-hexify-string (buffer-name))))

(defvar-local my/imp-preview-buffer nil
  "The xwidget buffer showing this buffer's preview, if any.")

(defun my/imp-show-preview ()
  "Open the current buffer's preview and return its xwidget buffer.
The xwidget is shown in a side window. Without xwidget support the
preview opens in the external browser and the return value is nil."
  (let ((source (current-buffer))
        (url (my/imp-live-url)))
    (if (not (featurep 'xwidget-internal))
        (progn (browse-url url) nil)
      ;; A new session per preview, so each source buffer gets its own.
      (xwidget-webkit-browse-url url t)
      (let ((preview (current-buffer)))
        (switch-to-buffer source)
        (display-buffer preview
                        '(display-buffer-in-side-window
                          (side . right)
                          (window-width . 0.5)))
        preview))))

(defun my/markdown-preview-toggle ()
  "Toggle the live HTML preview of the current markdown buffer."
  (interactive)
  (if impatient-mode
      (progn
        (impatient-mode -1)
        (when (buffer-live-p my/imp-preview-buffer)
          (let ((kill-buffer-query-functions nil))
            (kill-buffer my/imp-preview-buffer)))
        (setq my/imp-preview-buffer nil)
        (unless (seq-some (lambda (buf) (buffer-local-value 'impatient-mode buf))
                          (buffer-list))
          (httpd-stop))
        (message "Markdown preview off"))
    (unless (httpd-running-p)
      (httpd-start))
    (impatient-mode 1)
    (imp-set-user-filter #'my/imp-markdown-filter)
    (setq my/imp-preview-buffer (my/imp-show-preview))
    (message "Markdown preview on")))

** Semester calendar

Dates for the current semester, taken from the QUT academic calendar.
Update these four values at the start of each semester and the start
page works out the week by itself. Break weeks are skipped when
counting, so the number matches QUT's own week numbering.

#+begin_src emacs-lisp

(defvar my/qut-week-1 '(2026 7 27))
(defvar my/qut-break-weeks '((2026 9 7)))
(defvar my/qut-semester-end '(2026 11 14))
(defvar my/qut-teaching-weeks 13)

(defun my/qut--days (date)
  "Turn DATE, a (YEAR MONTH DAY) list, into an absolute day number."
  (pcase-let ((`(,y ,m ,d) date))
    (time-to-days (encode-time (list 0 0 12 d m y nil -1 nil)))))

(defun my/qut-week-label ()
  "Describe where today sits in the semester, or nil outside it."
  (let* ((today (time-to-days (current-time)))
         (start (my/qut--days my/qut-week-1))
         (breaks (mapcar #'my/qut--days my/qut-break-weeks))
         (in-break (seq-find (lambda (b) (<= b today (+ b 6))) breaks))
         (breaks-before (seq-count (lambda (b) (< (+ b 6) today)) breaks))
         (week (- (1+ (/ (- today start) 7)) breaks-before)))
    (cond ((or (< today start)
               (> today (my/qut--days my/qut-semester-end)))
           nil)
          (in-break "mid semester break")
          ((<= week my/qut-teaching-weeks) (format "week %d" week))
          (t "study and exams"))))
#+end_src

** Banner shimmer

Each time the start page is drawn, a band of light sweeps across the
banner a couple of times, then the banner settles on its usual colour.

The banner text is tagged with a =my/enlight-banner= property in
=my/enlight-build=. After enlight inserts its content, every tagged
character gets its own overlay, and a timer recolours those overlays
about 30 times a second. Overlays sit on top of the text without
modifying it, so the read only buffer is never touched and finishing
is just deleting them. They also evaporate if enlight redraws the
buffer, which ends a shimmer that is still running.

The band's position comes from the time elapsed rather than a frame
counter, so its speed stays steady even when Emacs is busy at startup
and some frames run late.

#+begin_src emacs-lisp
(require 'color)
(require 'cl-lib)

(defvar my/enlight-shimmer-passes 2
  "How many times the band sweeps across the banner.")
(defvar my/enlight-shimmer-speed 40
  "Speed of the band, in columns per second.")
(defvar my/enlight-shimmer-radius 6
  "Half width of the band, in columns.")
(defvar my/enlight-shimmer-strength 0.85
  "How far the band's centre moves from the keyword colour toward
the text colour, from 0 to 1.")
(defvar my/enlight-shimmer-delay 0.3
  "Seconds to wait before starting, so the frame is on screen first.")

(defvar my/enlight--shimmer-timer nil)
(defvar my/enlight--shimmer-cells nil
  "Overlays on the banner, each paired with its column.")

(defun my/enlight--mix (from to amount)
  "Blend the RGB list FROM toward TO by AMOUNT, from 0 to 1, as hex."
  (apply #'color-rgb-to-hex
         (append (cl-mapcar (lambda (a b) (+ a (* amount (- b a)))) from to)
                 '(2))))

(defun my/enlight--band (distance)
  "Brightness of a column DISTANCE columns from the band's centre.
A cosine curve gives soft edges instead of a hard bar."
  (if (>= distance my/enlight-shimmer-radius)
      0
    (/ (1+ (cos (/ (* float-pi distance) my/enlight-shimmer-radius))) 2)))

(defun my/enlight--banner-cells ()
  "Overlay every visible banner character in the current buffer.
Return a list of (OVERLAY . COLUMN), with columns counted from the
banner's left edge."
  (let ((cells
         (save-excursion
           (cl-loop for pos from (point-min) below (point-max)
                    when (and (get-text-property pos 'my/enlight-banner)
                              (not (memq (char-after pos) '(?\s ?\n))))
                    collect (let ((ov (make-overlay pos (1+ pos))))
                              (overlay-put ov 'evaporate t)
                              (goto-char pos)
                              (cons ov (current-column)))))))
    (when cells
      (let ((left (apply #'min (mapcar #'cdr cells))))
        (dolist (cell cells)
          (setcdr cell (- (cdr cell) left)))))
    cells))

(defun my/enlight-shimmer-stop ()
  "Stop the shimmer and restore the banner's normal colour."
  (when my/enlight--shimmer-timer
    (cancel-timer my/enlight--shimmer-timer)
    (setq my/enlight--shimmer-timer nil))
  (mapc (lambda (cell) (delete-overlay (car cell))) my/enlight--shimmer-cells)
  (setq my/enlight--shimmer-cells nil))

(defun my/enlight--shimmer-frame (base text start)
  "Draw one frame, blending the BASE colour toward TEXT inside the band.
START is when the first pass began."
  (let* ((cells my/enlight--shimmer-cells)
         (span (+ 1 (apply #'max (mapcar #'cdr cells))
                  (* 2 my/enlight-shimmer-radius)))
         (travelled (max 0 (* my/enlight-shimmer-speed
                              (float-time (time-since start)))))
         (centre (- (mod travelled span) my/enlight-shimmer-radius)))
    (if (or (>= travelled (* span my/enlight-shimmer-passes))
            (not (overlay-buffer (caar cells))))
        (my/enlight-shimmer-stop)
      (pcase-dolist (`(,ov . ,col) cells)
        (overlay-put ov 'face
                     (list :foreground
                           (my/enlight--mix base text
                                            (* my/enlight-shimmer-strength
                                               (my/enlight--band
                                                (abs (- col centre)))))))))))

(defun my/enlight-shimmer ()
  "Sweep a band of light across the banner in the current buffer.
The band blends the keyword colour toward the default text colour.
Nothing happens when either colour is unknown, as on some terminals."
  (my/enlight-shimmer-stop)
  (let* ((base (color-name-to-rgb
                (face-foreground 'font-lock-keyword-face nil t)))
         (text (color-name-to-rgb (face-foreground 'default nil t))))
    (when (and base text)
      (setq my/enlight--shimmer-cells (my/enlight--banner-cells))
      (when my/enlight--shimmer-cells
        (setq my/enlight--shimmer-timer
              (run-with-timer my/enlight-shimmer-delay (/ 1.0 30)
                              #'my/enlight--shimmer-frame
                              base text
                              (time-add nil my/enlight-shimmer-delay)))))))

(add-hook 'enlight-after-insert-hook #'my/enlight-shimmer)

* Claude

** Ghostel terminal

Terminal backend for Claude sessions, built on Ghostty's VT engine. Its
native module is downloaded from the GitHub releases on first use.

=C-p= / =C-n= scroll the chat one step and =s-p= / =s-n= five steps,
while Claude's prompt history stays on the arrow keys. When the app
tracks the mouse (Claude Code's fullscreen renderer), its history lives
inside the app and the buffer holds only the visible rows, so scrolling
sends it wheel presses (button 4 up, 5 down) at the middle row. Claude
does not use the alternate screen here, so =ghostel-alt-screen-p= cannot
detect this. Otherwise the history is buffer text, so the Emacs window
scrolls, and ghostel re-anchors it to the live cursor once it is back at
the bottom.

#+begin_src emacs-lisp
(use-package ghostel
  :defer t
  :custom
  (ghostel-module-auto-install 'download)
  :config
  (defun my/ghostel-scroll (button n)
    "Scroll N steps up (BUTTON 4) or down (BUTTON 5).
A step is a wheel press when the app tracks the mouse, else a window line."
    (if (ghostel--mouse-tracking-p ghostel--term)
        (dotimes (_ n)
          (ghostel--mouse-event ghostel--term 0 button
                                (/ (window-body-height) 2) 0 0))
      (if (= button 4) (scroll-down n) (scroll-up n))))
  (defun my/ghostel-scroll-up ()
    (interactive)
    (my/ghostel-scroll 4 1))
  (defun my/ghostel-scroll-down ()
    (interactive)
    (my/ghostel-scroll 5 1))
  (defun my/ghostel-scroll-up-5 ()
    (interactive)
    (my/ghostel-scroll 4 5))
  (defun my/ghostel-scroll-down-5 ()
    (interactive)
    (my/ghostel-scroll 5 5))

  ;; As exceptions, C-p and C-n are not sent to the terminal and fall
  ;; through to `ghostel-mode-map', the parent of the input mode maps.
  ;; Bindings there survive because ghostel rebuilds only
  ;; `ghostel-semi-char-mode-map' (from this exception list).
  (customize-set-variable 'ghostel-keymap-exceptions
                          (append ghostel-keymap-exceptions '("C-p" "C-n")))
  (define-key ghostel-mode-map (kbd "C-p") #'my/ghostel-scroll-up)
  (define-key ghostel-mode-map (kbd "C-n") #'my/ghostel-scroll-down)
  ;; Super keys are never sent to the terminal, so no exception is needed.
  (define-key ghostel-mode-map (kbd "s-p") #'my/ghostel-scroll-up-5)
  (define-key ghostel-mode-map (kbd "s-n") #'my/ghostel-scroll-down-5)
  ;; Copy and Emacs modes also inherit `ghostel-mode-map'; keep line motion.
  (define-key ghostel-readonly-mode-map (kbd "C-p") #'previous-line)
  (define-key ghostel-readonly-mode-map (kbd "C-n") #'next-line)
  (define-key ghostel-readonly-mode-map (kbd "s-p") #'my/previous-line-5)
  (define-key ghostel-readonly-mode-map (kbd "s-n") #'my/next-line-5)
  ;; Sends M-DEL, which the Claude CLI treats as backward-kill-word.
  (define-key ghostel-mode-map (kbd "s-<backspace>")
              (lambda () (interactive) (ghostel-send-key "backspace" "meta"))))
#+end_src

** claude-code-ide

Claude Code sessions inside Emacs, opened from =C-c C-'= or =M-x claude=,
with the Emacs MCP tools enabled.

#+begin_src emacs-lisp
(use-package claude-code-ide
  :vc (:url "https://github.com/manzaltu/claude-code-ide.el" :rev :newest)
  :bind ("C-c C-'" . claude-code-ide-menu)
  :custom
  (claude-code-ide-terminal-backend 'ghostel)
  ;; Diffs get their own frame, so the Claude window stays put in the main one.
  (claude-code-ide-show-claude-window-in-ediff nil)
  :init
  (defalias 'claude #'claude-code-ide-menu)
  :config
  ;; The MCP HTTP server signals `json-rpc-error' without defining it, so its
  ;; own handler cannot catch it. Defining it makes the server reply with a
  ;; JSON-RPC error instead of crashing the process filter.
  (define-error 'json-rpc-error "JSON-RPC error" 'error)

  ;; Answer the resources and prompts list requests with empty collections
  ;; instead of "Method not found".
  (advice-add 'claude-code-ide-mcp-http-server--dispatch :around
              (lambda (orig method params)
                (pcase method
                  ("resources/list" '((resources . [])))
                  ("resources/templates/list" '((resourceTemplates . [])))
                  ("prompts/list" '((prompts . [])))
                  (_ (funcall orig method params)))))

  ;; Review diffs in a new frame. The openDiff handler builds ediff in the
  ;; selected frame, so selecting a fresh frame first is enough. The tab
  ;; switch is skipped because it would look for Claude's tab in the new
  ;; frame. The frame is tagged after the handler returns, since the handler
  ;; first cleans up any older diff with the same tab name, and it is deleted
  ;; if the handler exits any other way, including C-g.
  ;;
  ;; The diff frame fills the left half of the screen. Frame size parameters
  ;; measure the text area only, so the title bar and fringes are subtracted
  ;; to keep the minibuffer, where the accept prompt appears, on screen.
  ;; The frame it was opened from is remembered so focus can return there.
  (defun my/claude-diff-make-frame ()
    (pcase-let* ((`(,x ,y ,width ,height) (frame-monitor-workarea))
                 (frame (make-frame `((name . "Claude diff")
                                      (fullscreen . nil)
                                      (my/claude-diff-origin . ,(selected-frame))))))
      (set-frame-position frame x y)
      (set-frame-size frame
                      (- (/ width 2) (- (frame-outer-width frame) (frame-text-width frame)))
                      (- height (- (frame-outer-height frame) (frame-text-height frame)))
                      t)
      frame))

  ;; Claude writes accepted edits straight to disk, which leaves the file's
  ;; buffer stale. The handler would then stop on a "Reread from disk?"
  ;; prompt, so a stale buffer without unsaved edits is reverted first.
  (defun my/claude-diff-in-new-frame (orig arguments &optional session)
    (let ((buffer (find-buffer-visiting (alist-get 'old_file_path arguments)))
          (opened nil)
          frame)
      (when (and buffer
                 (not (buffer-modified-p buffer))
                 (not (verify-visited-file-modtime buffer)))
        (with-current-buffer buffer
          (revert-buffer t t t)))
      (setq frame (my/claude-diff-make-frame))
      (select-frame-set-input-focus frame)
      (unwind-protect
          (prog1 (let ((claude-code-ide-switch-tab-on-ediff nil))
                   (funcall orig arguments session))
            (setq opened t)
            (set-frame-parameter frame 'my/claude-diff
                                 (cons session (alist-get 'tab_name arguments))))
        (unless opened
          (my/claude-diff-close-frame frame)))))

  ;; Deleting a frame only changes the selected frame inside Emacs, and macOS
  ;; leaves the remaining frame unfocused with a hollow cursor, so focus is
  ;; handed back explicitly.
  (defun my/claude-diff-close-frame (frame)
    (let ((origin (frame-parameter frame 'my/claude-diff-origin)))
      (delete-frame frame)
      (when (frame-live-p origin)
        (select-frame-set-input-focus origin))))

  (defun my/claude-diff-frame (tab-name session)
    (seq-find (lambda (frame)
                (let ((tag (frame-parameter frame 'my/claude-diff)))
                  (and (eq (car tag) session) (equal (cdr tag) tab-name))))
              (frame-list)))

  ;; Quitting ediff with q leaves its control panel alive, so when Claude
  ;; then sends close_tab the handler quits ediff a second time, and that
  ;; quit splits the selected frame. Running close_tab with the diff frame
  ;; selected keeps those splits off the main frame.
  (defun my/claude-diff-close-in-frame (orig arguments &optional session)
    (if-let* ((frame (my/claude-diff-frame (alist-get 'tab_name arguments) session)))
        (with-selected-frame frame
          (funcall orig arguments session))
      (funcall orig arguments session)))

  ;; Every way a diff ends goes through cleanup-diff, last of all.
  (defun my/claude-diff-delete-frame (tab-name session)
    (when-let* ((frame (my/claude-diff-frame tab-name session)))
      (my/claude-diff-close-frame frame)))

  (advice-add 'claude-code-ide-mcp-handle-open-diff :around #'my/claude-diff-in-new-frame)
  (advice-add 'claude-code-ide-mcp-handle-close-tab :around #'my/claude-diff-close-in-frame)
  (advice-add 'claude-code-ide-mcp--cleanup-diff :after #'my/claude-diff-delete-frame)

  (claude-code-ide-emacs-tools-setup))
#+end_src
