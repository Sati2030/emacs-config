** Remote (TRAMP) debugging

=my/tramp-debug-toggle= switches TRAMP logging between level 3 and
level 6. Level 6 logs all remote traffic and mirrors each
=*debug tramp/...*= buffer to a file under =temporary-file-directory=.
Toggling closes cached connections.

#+begin_src emacs-lisp
(setq tramp-verbose 3
      tramp-debug-to-file nil)

(defun my/tramp-debug-toggle ()
  "Toggle verbose TRAMP logging to debug buffers and to disk.
Closes cached connections."
  (interactive)
  (require 'tramp)
  (setq tramp-verbose (if (> tramp-verbose 3) 3 6)
        tramp-debug-to-file (> tramp-verbose 3))
  (tramp-cleanup-all-connections)
  (message "TRAMP verbose %d, file logging %s (%s)"
           tramp-verbose
           (if tramp-debug-to-file "on" "off")
           temporary-file-directory))
#+end_src

(defun my/esp-project-p ()
  "Return non-nil if the current buffer is inside an ESP-IDF or PlatformIO project."
  (let ((dir (or buffer-file-name default-directory)))

*** Reload PlatformIO compile_commands.json for clangd

=my/pio-reload-compile-commands= regenerates =compile_commands.json=
with =pio run -t compiledb= for the =default_envs= of =platformio.ini=,
then restarts the LSP workspace. With =C-u= it prompts for an
=[env:NAME]= instead.

#+begin_src emacs-lisp
(defun my/pio-envs ()
  "Return the env names declared as [env:NAME] in platformio.ini."
  (with-temp-buffer
    (insert-file-contents (expand-file-name "platformio.ini" (my/project-root)))
    (let (envs)
      (while (re-search-forward "^\\[env:\\([^]]+\\)\\]" nil t)
        (push (match-string 1) envs))
      (nreverse envs))))

(defun my/pio-reload-compile-commands (&optional env)
  "Regenerate compile_commands.json via PlatformIO, then restart the LSP workspace.
Builds the default_envs of platformio.ini, or ENV when given. With a
prefix argument, asks for ENV. Runs pio in a hidden buffer that is
shown only if the build fails."
  (interactive
   (list (and current-prefix-arg
              (completing-read "PlatformIO env: " (my/pio-envs) nil t))))
  (let* ((default-directory (my/project-root))
         ;; LSP workspaces of the calling buffer.
         (workspaces (and (fboundp 'lsp-workspaces) (lsp-workspaces)))
         ;; Leading space hides the buffer from buffer lists.
         (buf (get-buffer-create " *pio-compiledb*")))
    (when (process-live-p (get-buffer-process buf))
      (user-error "pio compiledb is already running"))
    (with-current-buffer buf (erase-buffer))
    (message "Running pio compiledb...")
    (make-process
     :name "pio-compiledb"
     :buffer buf
     :command (append '("pio" "run" "-t" "compiledb")
                      (and env (list "-e" env)))
     :sentinel
     (lambda (process _event)
       (when (memq (process-status process) '(exit signal))
         (if (zerop (process-exit-status process))
             (progn
               (message "pio compiledb done, restarting LSP")
               (mapc #'lsp-workspace-restart workspaces))
           (message "pio compiledb failed")
           (display-buffer (process-buffer process))))))))
#+end_src

** C#

*** Brace auto-formatting

Typing ={= opens an indented block on a new line, and =}= closes it,
leaving point on the empty line between the braces. Inside strings, and
after a space for ={= or after code for =}=, the brace is inserted as is.

#+begin_src emacs-lisp
(defun my/in-string-p ()
  "Return non-nil if point is inside a string."
  (nth 3 (syntax-ppss)))

(defun my/csharp-open-brace ()
  "Insert {, opening an indented block unless in a string or after a space."
  (interactive)
  (if (or (my/in-string-p)
          (eq (char-before) ?\s))
      (insert "{")
    (newline)
    (insert "{")
    (indent-according-to-mode)
    (newline)
    (indent-according-to-mode)))

(defun my/csharp-close-brace ()
  "Insert }, closing an indented block unless in a string or after code."
  (interactive)
  (if (or (my/in-string-p)
          (and (not (memq (char-before) '(?\s ?\t)))
               (save-excursion
                 (search-backward "{" nil t))))
      (insert "}")
    (newline)
    (insert "}")
    (indent-according-to-mode)
    (forward-line -1)
    (indent-according-to-mode)))

(add-hook 'csharp-mode-hook
          (lambda ()
            (local-set-key (kbd "{") #'my/csharp-open-brace)
            (local-set-key (kbd "}") #'my/csharp-close-brace)))
#+end_src

*** Live preview via impatient-mode + pandoc

=C-c C-c l= (=my/markdown-preview-toggle=) serves the buffer through
impatient-mode as HTML rendered by pandoc with MathJax, shown in an
=xwidget-webkit= side window (or the external browser without xwidget
support). Toggling off stops serving the buffer, closes the window, and
stops the server once no buffer is served.

pandoc runs asynchronously after =my/imp-render-delay= idle seconds,
replacing impatient-mode's change hook. One render runs per buffer; an
edit made during a render reruns it when it finishes. Uses the
impatient-mode internals =imp--on-change= and =imp--update-buffer=.

#+begin_src emacs-lisp
(use-package simple-httpd)
(use-package impatient-mode)

(defun my/browse-url-in-emacs (url)
  "Open URL in an xwidget-webkit buffer, or externally without xwidget support."
  (if (featurep 'xwidget-internal)
      (xwidget-webkit-browse-url url)
    (browse-url url)))

(defvar my/imp-render-delay 0.2
  "Idle seconds after an edit before the preview is rendered again.")

(defvar-local my/imp--html ""
  "HTML of the last finished render of this buffer.")
(defvar-local my/imp--process nil
  "The pandoc process rendering this buffer, if one is running.")
(defvar-local my/imp--stale nil
  "Non-nil if the buffer changed while pandoc was rendering it.")
(defvar-local my/imp--timer nil
  "Idle timer for the next render, if one is scheduled.")

(defun my/imp-markdown-filter (buffer)
  "Print the HTML of BUFFER's last finished render."
  (princ (buffer-local-value 'my/imp--html buffer)))

(defun my/imp--render ()
  "Render the current buffer to HTML with pandoc in the background.
If a render is already running, mark it stale so it reruns when done."
  (if (process-live-p my/imp--process)
      (setq my/imp--stale t)
    (let* ((source (current-buffer))
           (output (generate-new-buffer " *pandoc*"))
           (errors (generate-new-buffer " *pandoc errors*"))
           (process
            (make-process
             :name "pandoc"
             :buffer output
             :stderr errors
             :connection-type 'pipe
             :coding 'utf-8
             :noquery t
             :command '("pandoc" "-f" "markdown+tex_math_dollars"
                        "-t" "html5" "--standalone" "--mathjax")
             :sentinel (lambda (process _event)
                         (my/imp--render-done process source output errors)))))
      (setq my/imp--stale nil
            my/imp--process process)
      (save-restriction
        (widen)
        (process-send-region process (point-min) (point-max)))
      (process-send-eof process))))

(defun my/imp--render-done (process source output errors)
  "Push PROCESS's HTML in OUTPUT to SOURCE's preview, then clean up.
On failure the page keeps the last good render and the first line of
ERRORS is shown."
  (when (memq (process-status process) '(exit signal))
    (when (and (buffer-live-p source)
               (eq process (buffer-local-value 'my/imp--process source)))
      (with-current-buffer source
        (setq my/imp--process nil)
        (if (zerop (process-exit-status process))
            (progn
              (setq my/imp--html (with-current-buffer output (buffer-string)))
              (imp--update-buffer))
          (message "pandoc failed: %s"
                   (with-current-buffer errors
                     (buffer-substring (point-min)
                                       (save-excursion
                                         (goto-char (point-min))
                                         (line-end-position))))))
        (when my/imp--stale
          (my/imp--render))))
    (kill-buffer output)
    (kill-buffer errors)))

(defun my/imp--schedule-render (&rest _)
  "Render the current buffer once Emacs has been idle for a moment.
Meant for `after-change-functions'."
  (unless my/imp--timer
    (setq my/imp--timer
          ;; Count the delay from now, even if Emacs is already idle.
          (run-with-idle-timer
           (time-add (or (current-idle-time) 0) my/imp-render-delay)
           nil #'my/imp--scheduled-render (current-buffer)))))

(defun my/imp--scheduled-render (buffer)
  "Run BUFFER's scheduled render."
  (when (buffer-live-p buffer)
    (with-current-buffer buffer
      (setq my/imp--timer nil)
      (my/imp--render))))

(defun my/imp--stop-rendering ()
  "Cancel the current buffer's pending render and kill a running one."
  (remove-hook 'after-change-functions #'my/imp--schedule-render t)
  (when my/imp--timer
    (cancel-timer my/imp--timer)
    (setq my/imp--timer nil))
  ;; Cleared before killing, so the sentinel ignores the process.
  (let ((process my/imp--process))
    (setq my/imp--process nil
          my/imp--stale nil
          my/imp--html "")
    (when (process-live-p process)
      (delete-process process))))

(defun my/imp-live-url ()
  "URL that impatient-mode serves the current buffer at."
  (format "http://localhost:%d/imp/live/%s/"
          httpd-port (url-hexify-string (buffer-name))))

(defvar-local my/imp-preview-buffer nil
  "The xwidget buffer showing this buffer's preview, if any.")

(defun my/show-url-in-side-window (url)
  "Show URL in a right side window and return its xwidget buffer.
Without xwidget support URL opens in the external browser and the
return value is nil. The selected window and current buffer are left
as they were."
  (if (not (featurep 'xwidget-internal))
      (progn (browse-url url) nil)
    (let ((window (selected-window))
          (shown (window-buffer)))
      (save-current-buffer
        ;; New session, so each preview gets its own buffer.
        (xwidget-webkit-browse-url url t)
        (let ((preview (current-buffer)))
          ;; Restore the selected window's buffer.
          (set-window-buffer window shown)
          (display-buffer preview
                          '(display-buffer-in-side-window
                            (side . right)
                            (window-width . 0.5)))
          preview)))))

(defun my/imp-close-preview ()
  "Stop serving the current buffer and close its preview.
The server stops too once no buffer is being served."
  (impatient-mode -1)
  (when (buffer-live-p my/imp-preview-buffer)
    (let ((kill-buffer-query-functions nil))
      (kill-buffer my/imp-preview-buffer)))
  (setq my/imp-preview-buffer nil)
  (unless (seq-some (lambda (buf) (buffer-local-value 'impatient-mode buf))
                    (buffer-list))
    (httpd-stop)))

(defun my/markdown-preview-toggle ()
  "Toggle the live HTML preview of the current markdown buffer."
  (interactive)
  (if impatient-mode
      (progn
        (my/imp--stop-rendering)
        (my/imp-close-preview)
        (message "Markdown preview off"))
    (unless (httpd-running-p)
      (httpd-start))
    (impatient-mode 1)
    (remove-hook 'after-change-functions #'imp--on-change t)
    (add-hook 'after-change-functions #'my/imp--schedule-render nil t)
    (imp-set-user-filter #'my/imp-markdown-filter)
    (my/imp--render)
    (setq my/imp-preview-buffer (my/show-url-in-side-window (my/imp-live-url)))
    (message "Markdown preview on")))

** Indenting code inside JavaScript template literals

Indents lines inside multi-line template literals (such as WebGL
shaders) by brace counting, before the mode's own indenter:

- base column is the indentation of the line holding the opening backtick
- each unclosed ={= inside the literal adds one =js-indent-level=
- a line starting with =}= sits one level shallower
- the line with the closing backtick returns to the base column

Other lines use the normal JS indenter.

#+begin_src emacs-lisp
(defun my/js-template-literal-indent-line ()
  "Indent the current line if it starts inside a template literal.
Return non-nil when the line was handled, nil to defer to the mode."
  (let* ((ppss (save-excursion (syntax-ppss (line-beginning-position))))
         (open (nth 8 ppss)))
    (when (and (nth 3 ppss) (eq (char-after open) ?`))
      (let* ((step (if (boundp 'js-indent-level) js-indent-level 4))
             (base (save-excursion (goto-char open) (current-indentation)))
             (first-char (save-excursion (back-to-indentation) (char-after)))
             (bol (line-beginning-position))
             (depth 0))
        ;; Net brace depth between the opening backtick and this line.
        (save-excursion
          (goto-char (1+ open))
          (while (progn (skip-chars-forward "^{}" bol) (< (point) bol))
            (setq depth (+ depth (if (eq (char-after) ?{) 1 -1)))
            (forward-char 1)))
        (when (eq first-char ?}) (setq depth (1- depth)))
        (let ((target (if (eq first-char ?`)
                          base
                        (+ base (* step (1+ (max 0 depth))))))
              (offset (- (point) (save-excursion (back-to-indentation) (point)))))
          ;; Indent with spaces only.
          (let ((indent-tabs-mode nil))
            (indent-line-to target))
          (when (> offset 0) (forward-char offset))))
      t)))

(defun my/js-enable-template-literal-indent ()
  "Try the template literal indenter before the mode's own one."
  (add-function :before-until (local 'indent-line-function)
                #'my/js-template-literal-indent-line))

(add-hook 'js-mode-hook    #'my/js-enable-template-literal-indent)
(add-hook 'js-ts-mode-hook #'my/js-enable-template-literal-indent)

** Web frontend previews

*** Static pages via live-server

=M-x my/frontend-preview= toggles a dev server in the project root and
shows the page in an =xwidget-webkit= side window (or the external
browser). It runs =live-server --port=0= and opens the current file
when it is an HTML page, else the root's =index.html=. Killing
=*live-server*= also stops it. Page console output goes to
=*frontend-console*= (see "Browser console in Emacs" below). The page
opens when the server prints its "Serving ... at URL" line. Requires:

#+begin_example
npm i -g live-server
#+end_example

When =package.json= has a =dev= script, =npm run dev= runs instead and
the root URL opens. The script must honor the =PORT= environment
variable (=0= for any free port) and print =Serving <something> at
<URL>= once listening. Stopping kills the server's whole process
group.

#+begin_src emacs-lisp
(require 'ansi-color)
(require 'subr-x)                       ; string-trim, string-remove-suffix

(defvar my/live-server-process nil
  "The running live-server process, if any.")

(defun my/npm-dev-script-p (root)
  "Non-nil when ROOT has a package.json defining a `dev' script."
  (let ((file (expand-file-name "package.json" root)))
    (and (file-readable-p file)
         (let ((json (json-parse-string
                      (with-temp-buffer
                        (insert-file-contents file)
                        (buffer-string))
                      :object-type 'alist)))
           (alist-get 'dev (alist-get 'scripts json))))))

(defun my/frontend-preview ()
  "Toggle a dev server for the current project, shown in a side window.
Runs `npm run dev' when the project defines that script, else
live-server. With live-server, opens the current file when it is an
HTML page, else the root's index."
  (interactive)
  (if (process-live-p my/live-server-process)
      (progn
        (delete-process my/live-server-process)
        (message "Frontend preview off"))
    (let* ((root (my/project-root))
           (default-directory root)
           (npm (my/npm-dev-script-p root))
           (page (and (not npm)
                      buffer-file-name
                      (string-match-p "\\.html?\\'" buffer-file-name)
                      (file-relative-name buffer-file-name root)))
           (process-environment (cons "PORT=0" process-environment))
           (output (get-buffer-create "*live-server*")))
      (when (and npm (not (file-directory-p
                           (expand-file-name "node_modules" root))))
        (message "node_modules missing, running npm install first...")
        (call-process "npm" nil "*npm-install*" t "install"))
      (with-current-buffer output
        (let ((inhibit-read-only t))
          (erase-buffer)))
      (setq my/live-server-process
            (make-process
             :name (if npm "npm-dev" "live-server")
             :buffer output
             :command (if npm
                          '("npm" "run" "dev")
                        '("live-server" "--no-browser" "--port=0"))
             :connection-type 'pipe
             :noquery t
             :filter (lambda (process string)
                       (my/live-server--filter process string page))
             :sentinel #'my/live-server--sentinel))
      (message "Starting %s in %s ..."
               (if npm "npm run dev" "live-server") root))))

(defun my/frontend-preview--page-url (base page)
  "URL of PAGE under the server at BASE.
PAGE is a path relative to the served root, or nil for the index."
  (concat (string-remove-suffix "/" base) "/"
          (and page (mapconcat #'url-hexify-string
                               (split-string page "/")
                               "/"))))

(defun my/live-server--filter (process string page)
  "Log PROCESS's STRING and act on each \"Serving ... at URL\" line.
PAGE is a path relative to the served root, or nil for the index."
  (let ((log (process-buffer process)))
    (when (buffer-live-p log)
      (with-current-buffer log
        (save-excursion
          (goto-char (point-max))
          (insert (ansi-color-filter-apply string)))))
    ;; Scan complete lines; keep the unfinished tail for the next call.
    (let ((lines (split-string
                  (concat (or (process-get process 'partial) "") string)
                  "\n")))
      (process-put process 'partial (car (last lines)))
      (dolist (line (butlast lines))
        (let ((line (ansi-color-filter-apply line)))
          (when (string-match "Serving .* at \\(https?://[^[:space:]]+\\)"
                              line)
            (my/frontend-preview--server-ready
             process (match-string 1 line) page)))))))

(defun my/frontend-preview--server-ready (server url page)
  "Handle SERVER announcing that it listens at URL; PAGE is as above.
The first announcement opens the preview, through the console proxy
when available. Later ones repoint the proxy."
  (let ((proxy (process-get server 'proxy)))
    (cond
     ((process-live-p proxy)
      (process-send-string proxy (concat url "\n")))
     ((process-get server 'preview-opened))
     (t
      (process-put server 'preview-opened t)
      (if (my/frontend-console-available-p)
          (process-put server 'proxy
                       (my/frontend-console--start server url page))
        (process-put server 'preview
                     (my/show-url-in-side-window
                      (my/frontend-preview--page-url url page))))))))

(defun my/live-server--sentinel (process event)
  "Close PROCESS's proxy and preview once the dev server has exited.
Reports EVENT when the server exits on its own rather than being stopped."
  (unless (process-live-p process)
    (let ((proxy (process-get process 'proxy)))
      (when (process-live-p proxy)
        (delete-process proxy)))
    (let ((preview (process-get process 'preview)))
      (when (buffer-live-p preview)
        (let ((kill-buffer-query-functions nil))
          (kill-buffer preview))))
    (when (eq process my/live-server-process)
      (setq my/live-server-process nil))
    (when (eq (process-status process) 'exit)
      (message "%s %s (see *live-server*)"
               (process-name process) (string-trim event)))))
#+end_src

*** Browser console in Emacs

=lisp/frontend-console-proxy.js= (Node, no dependencies) is a reverse
proxy between the preview and the dev server. It forwards requests
unchanged and injects =lisp/frontend-console-client.js= at the top of
each HTML =<head>=. That script reports console calls, uncaught errors,
unhandled rejections and failed resource loads, which the proxy prints
as =[level] text= lines in =*frontend-console*=. Each reload starts with
a =[page] loaded URL= line. WebSocket upgrades pass through unchanged.
When the dev server restarts on a new port, Emacs sends the new URL to
the proxy's stdin.

Without =node=, the preview opens the server directly, with no console
buffer. Killing =*frontend-console*= kills the proxy and breaks the
preview; stop it with =my/frontend-preview= instead.

#+begin_src emacs-lisp
(defconst my/frontend-console-proxy-script
  (expand-file-name "lisp/frontend-console-proxy.js" user-emacs-directory)
  "Node script that proxies the dev server and collects browser logs.")

(defvar my/frontend-console-font-lock-keywords
  '(("^\\[error\\].*" . 'error)
    ("^\\[warn\\].*" . 'warning)
    ("^\\[page\\].*" . 'success)
    ("^\\[\\(?:debug\\|proxy\\)\\].*" . 'shadow))
  "Colors for *frontend-console* lines, keyed on their level prefix.")

(define-derived-mode my/frontend-console-mode special-mode "Browser-Console"
  "Major mode for the browser console log of `my/frontend-preview'."
  (setq-local font-lock-defaults '(my/frontend-console-font-lock-keywords t)))

(defun my/frontend-console-available-p ()
  "Non-nil when the console proxy can run."
  (and (executable-find "node")
       (file-readable-p my/frontend-console-proxy-script)))

(defun my/frontend-console--start (server url page)
  "Start the console proxy in front of SERVER, listening at URL.
The preview of PAGE opens once the proxy is ready. Returns the process."
  (let ((buffer (get-buffer-create "*frontend-console*")))
    (with-current-buffer buffer
      (my/frontend-console-mode)
      (let ((inhibit-read-only t))
        (erase-buffer)))
    (make-process
     :name "frontend-console"
     :buffer buffer
     :command (list "node" my/frontend-console-proxy-script url)
     :connection-type 'pipe
     :noquery t
     :filter (lambda (process string)
               (my/frontend-console--filter process string server page))
     :sentinel #'my/frontend-console--sentinel)))

(defun my/frontend-console--filter (process string server page)
  "Log PROCESS's STRING, opening SERVER's preview of PAGE once ready.
Windows showing the end of the log follow new output, like a terminal."
  (let ((buffer (process-buffer process)))
    (when (buffer-live-p buffer)
      (with-current-buffer buffer
        (let ((following
               (seq-filter (lambda (window)
                             (= (window-point window) (point-max)))
                           (get-buffer-window-list buffer nil t)))
              (inhibit-read-only t))
          (save-excursion
            (goto-char (point-max))
            (insert string)
            ;; Search the whole log for the ready line.
            (unless (process-get process 'preview-opened)
              (goto-char (point-min))
              (when (re-search-forward
                     "^Proxy for .* at \\(https?://[^[:space:]]+\\)" nil t)
                (process-put process 'preview-opened t)

  "Describe where today sits in the semester, or nil outside it."

         (start (my/qut--days my/qut-week-1))
         (breaks (mapcar #'my/qut--days my/qut-break-weeks))
         (in-break (seq-find (lambda (b) (<= b today (+ b 6))) breaks))

               (> today (my/qut--days my/qut-semester-end)))
           nil)
          (in-break "mid semester break")

** Banner shimmer

A band of light sweeps across the banner every few seconds while the
start page is in the selected window, or the minibuffer was opened from
it. The loop stops when another window is selected or the page is
hidden, and buffer local =window-buffer-change-functions= and
=window-selection-change-functions= hooks restart it.

Each banner character (tagged =my/enlight-banner= in =my/enlight-build=)
gets an overlay, recoloured =my/enlight-shimmer-fps= times a second. The
band's position comes from elapsed time. Brightness is rounded to
=my/enlight-shimmer-levels= prebuilt faces, an overlay is updated only
when its step changes, and frames are skipped while input is pending.

#+begin_src emacs-lisp
(require 'color)
(require 'cl-lib)

(defvar my/enlight-shimmer-fps 30
  "Frames drawn per second.")
(defvar my/enlight-shimmer-rest 4
  "Seconds of rest between sweeps.")
(defvar my/enlight-shimmer-speed 40
  "Speed of the band, in columns per second.")
(defvar my/enlight-shimmer-radius 6
  "Half width of the band, in columns.")
(defvar my/enlight-shimmer-strength 0.85
  "How far the band's centre moves from the keyword colour toward
the text colour, from 0 to 1.")
(defvar my/enlight-shimmer-levels 16
  "Number of brightness steps between the keyword and peak colour.")
(defvar my/enlight-shimmer-delay 0.3
  "Seconds to wait before starting, so the frame is on screen first.")

(defvar my/enlight--shimmer-timer nil)
(defvar my/enlight--shimmer-cells nil
  "One [OVERLAY COLUMN STEP] vector per banner character.")
(defvar my/enlight--shimmer-centre nil
  "Band centre of the last frame drawn, to skip identical frames.")

(defun my/enlight--mix (from to amount)
  "Blend the RGB list FROM toward TO by AMOUNT, from 0 to 1, as hex."
  (apply #'color-rgb-to-hex
         (append (cl-mapcar (lambda (a b) (+ a (* amount (- b a)))) from to)
                 '(2))))

(defun my/enlight--shimmer-faces (base text)
  "Build one face per brightness step, blending BASE toward TEXT.
Step 0 is nil, so the banner's own face shows through."
  (let ((faces (make-vector (1+ my/enlight-shimmer-levels) nil)))
    (dotimes (i my/enlight-shimmer-levels)
      (let ((step (1+ i)))
        (aset faces step
              (list :foreground
                    (my/enlight--mix base text
                                     (* my/enlight-shimmer-strength
                                        (/ (float step)
                                           my/enlight-shimmer-levels)))))))
    faces))

(defun my/enlight--band (distance)
  "Brightness of a column DISTANCE columns from the band's centre.
Falls off along a cosine curve."
  (if (>= distance my/enlight-shimmer-radius)
      0
    (/ (1+ (cos (/ (* float-pi distance) my/enlight-shimmer-radius))) 2)))

(defun my/enlight--banner-cells ()
  "Overlay every visible banner character in the current buffer.
Return a list of [OVERLAY COLUMN STEP] vectors, with columns counted
from the banner's left edge and every step starting at 0."
  (let ((cells
         (save-excursion
           (cl-loop for pos from (point-min) below (point-max)
                    when (and (get-text-property pos 'my/enlight-banner)
                              (not (memq (char-after pos) '(?\s ?\n))))
                    collect (let ((ov (make-overlay pos (1+ pos))))
                              (overlay-put ov 'evaporate t)
                              (goto-char pos)
                              (vector ov (current-column) 0))))))
    (when cells
      (let ((left (apply #'min (mapcar (lambda (c) (aref c 1)) cells))))
        (dolist (cell cells)
          (aset cell 1 (- (aref cell 1) left)))))
    cells))

(defun my/enlight-shimmer-stop ()
  "Stop the shimmer and restore the banner's normal colour."
  (when my/enlight--shimmer-timer
    (cancel-timer my/enlight--shimmer-timer)
    (setq my/enlight--shimmer-timer nil))
  (mapc (lambda (cell) (delete-overlay (aref cell 0))) my/enlight--shimmer-cells)
  (setq my/enlight--shimmer-cells nil
        my/enlight--shimmer-centre nil))

(defun my/enlight--shimmer-selected-p (buffer)
  "Non-nil when BUFFER is live and shown in the selected window.
When the minibuffer is active, the window it was opened from counts."
  (and buffer
       (eq (window-buffer (if (minibufferp (window-buffer (selected-window)))
                              (minibuffer-selected-window)
                            (selected-window)))
           buffer)))

(defun my/enlight--shimmer-frame (faces span cycle start)
  "Draw one frame using the step FACES.
SPAN is how many columns one sweep covers, CYCLE the seconds of one
sweep plus its rest, and START when the first sweep began."
  (let ((cells my/enlight--shimmer-cells))
    (cond
     ((not (my/enlight--shimmer-selected-p (overlay-buffer (aref (car cells) 0))))
      (my/enlight-shimmer-stop))
     ((input-pending-p))
     (t
      (let* ((elapsed (max 0 (float-time (time-since start))))
             (centre (- (min span (* my/enlight-shimmer-speed (mod elapsed cycle)))
                        my/enlight-shimmer-radius)))
        (unless (eql centre my/enlight--shimmer-centre)
          (setq my/enlight--shimmer-centre centre)
          (dolist (cell cells)
            (let ((step (round (* my/enlight-shimmer-levels
                                  (my/enlight--band
                                   (abs (- (aref cell 1) centre)))))))
              (unless (eql step (aref cell 2))
                (aset cell 2 step)
                (overlay-put (aref cell 0) 'face (aref faces step)))))))))))

(defun my/enlight-shimmer ()
  "Loop a band of light across the banner in the current buffer.
The band blends the keyword colour toward the default text colour.
Nothing happens when either colour is unknown, as on some terminals."
  (my/enlight-shimmer-stop)
  (add-hook 'window-buffer-change-functions #'my/enlight--shimmer-resume nil t)
  (add-hook 'window-selection-change-functions #'my/enlight--shimmer-resume nil t)
  (let* ((base (color-name-to-rgb
                (face-foreground 'font-lock-keyword-face nil t)))
         (text (color-name-to-rgb (face-foreground 'default nil t))))
    (when (and base text)
      (setq my/enlight--shimmer-cells (my/enlight--banner-cells))
      (when my/enlight--shimmer-cells
        (let* ((width (1+ (apply #'max (mapcar (lambda (c) (aref c 1))
                                               my/enlight--shimmer-cells))))
               (span (+ width (* 2 my/enlight-shimmer-radius)))
               (cycle (+ (/ (float span) my/enlight-shimmer-speed)
                         my/enlight-shimmer-rest)))
          (setq my/enlight--shimmer-timer
                (run-with-timer my/enlight-shimmer-delay
                                (/ 1.0 my/enlight-shimmer-fps)
                                #'my/enlight--shimmer-frame
                                (my/enlight--shimmer-faces base text)
                                span cycle
                                (time-add nil my/enlight-shimmer-delay))))))))

(defun my/enlight--shimmer-resume (window)
  "Restart the shimmer when WINDOW, showing the start page, is selected."
  (when (and (not my/enlight--shimmer-timer)
             (eq window (selected-window)))
    (with-current-buffer (window-buffer window)
      (when (derived-mode-p 'enlight-mode)
        (my/enlight-shimmer)))))

(add-hook 'enlight-after-insert-hook #'my/enlight-shimmer)
#+end_src

** Matrix rain

Green falling code drawn into the empty parts of the start page,
behind the dashboard. Each frame updates overlays without modifying the
buffer:

- runs of spaces get a same-width =display= string of rain characters,
- a =before-string= past each line's end fills the row to the window's
  edge, and the last line's one also fills the rows below,
- enlight's centring =line-prefix= is turned off and replaced by margin
  strings of the same width.

Gaps narrower than =my/enlight-rain-min-gap=, and the gap before a menu
key, stay blank.

Each column holds one drop with its own speed and a fading trail. A
drop that falls off the bottom restarts above the top after a random
pause. Cells keep their character between frames, and a share of them
(=my/enlight-rain-glitch=) changes each frame.

Overlays are rebuilt when the window size changes. The loop stops,
keeping the last frame, when the page is not in the selected window.
Only every =my/enlight-rain-column-step= columns carries rain, and the
per-frame functions are byte compiled at load.

#+begin_src emacs-lisp
(defvar my/enlight-rain-enabled t
  "Non-nil means draw falling code behind the start page.")
(defvar my/enlight-rain-fps 15
  "Frames per second.")
(defvar my/enlight-rain-speed '(6 . 18)
  "Slowest and fastest drop speeds, in rows per second.")
(defvar my/enlight-rain-colour "#00ff41"
  "Colour of the trails.")
(defvar my/enlight-rain-head-colour "#d8ffd8"
  "Colour of the character leading each trail.")
(defvar my/enlight-rain-strength 0.5
  "How far a trail's brightest shade moves from the background toward
`my/enlight-rain-colour', from 0 to 1. Lower keeps the page readable.")
(defvar my/enlight-rain-head-strength 0.8
  "Like `my/enlight-rain-strength', for the leading character.")
(defvar my/enlight-rain-levels 8
  "Number of shades a trail fades through.")
(defvar my/enlight-rain-glitch 0.01
  "Share of the characters swapped every frame.")
(defvar my/enlight-rain-min-gap 4
  "Gaps inside a line narrower than this stay blank.")
(defvar my/enlight-rain-column-step 2
  "Only every this many columns carries rain.")
(defvar my/enlight-rain-chars
  "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ$+-*/=%\"'#&_(),.;:?!|{}<>[]^~"
  "Characters the rain is made of. Every one must be one column wide.")
(defvar my/enlight-rain-delay 0.3
  "Seconds to wait before starting, so the frame is on screen first.")

(defvar my/enlight--rain-timer nil)
(defvar my/enlight--rain-buffer nil)
(defvar my/enlight--rain-overlays nil
  "Every overlay the rain owns.")
(defvar my/enlight--rain-segments nil
  "Lists of (OVERLAY PROPERTY ROW COLUMN WIDTH BELOW). The OVERLAY's
PROPERTY shows WIDTH columns of ROW from COLUMN, then BELOW whole rows
under it.")
(defvar my/enlight--rain-size nil
  "(COLUMNS . ROWS) of the window the overlays were built for.")
(defvar my/enlight--rain-grid nil
  "Character of every cell, row after row.")
(defvar my/enlight--rain-drops nil
  "One [HEAD SPEED LENGTH] vector per column. HEAD is a row, and may
be fractional or above the top edge.")
(defvar my/enlight--rain-faces nil
  "Trail shades, dimmest first.")
(defvar my/enlight--rain-head-face nil)
(defvar my/enlight--rain-time nil
  "When the last frame was drawn.")

(defun my/enlight--rain-random-char ()
  (aref my/enlight-rain-chars (random (length my/enlight-rain-chars))))

(defun my/enlight--rain-new-drop (rows &optional anywhere)
  "A drop for a screen ROWS tall, waiting above the top edge.
With ANYWHERE it may start part way down instead, so the first frame
is already full of rain."
  (pcase-let ((`(,slow . ,fast) my/enlight-rain-speed))
    (vector (float (- (random (* rows (if anywhere 2 1))) rows))
            (+ slow (* (- fast slow) (/ (random 1000) 1000.0)))
            (+ 4 (random (max 1 (/ rows 2)))))))

(defun my/enlight--rain-make-faces ()
  "Compute the shades from the current theme.
Return nil when the background colour is unknown, as on some
terminals."
  (let ((bg (color-name-to-rgb (face-background 'default nil t)))
        (rain (color-name-to-rgb my/enlight-rain-colour))
        (head (color-name-to-rgb my/enlight-rain-head-colour)))
    (when (and bg rain head)
      (setq my/enlight--rain-head-face
            (list :foreground
                  (my/enlight--mix bg head my/enlight-rain-head-strength))
            my/enlight--rain-faces
            (vconcat
             (cl-loop for i from 1 to my/enlight-rain-levels
                      collect (list :foreground
                                    (my/enlight--mix
                                     bg rain
                                     (* my/enlight-rain-strength
                                        (/ (float i) my/enlight-rain-levels)))))))
      t)))

(defun my/enlight--rain-line-segments (row margin cols)
  "Make the overlays for the line at point, shown on screen ROW.
MARGIN is the width of the left margin and COLS the window's width.
Return the segments with the line's last one first."
  (let* ((bol (line-beginning-position))
         (eol (line-end-position))
         (edge (1- cols))
         (runs (save-excursion
                 (cl-loop while (re-search-forward " +" eol t)
                          collect (cons (match-beginning 0) (match-end 0)))))
         (column (lambda (pos) (+ margin (- pos bol))))
         segments)
    (cl-flet ((add (start end prop col width)
                ;; Width clamped to the window's last column.
                (let ((ov (make-overlay start end)))
                  (push ov my/enlight--rain-overlays)
                  (push (list ov prop row col
                              (max 0 (min width (- edge col)))
                              0)
                        segments))))
      (if (or (= bol eol) (equal runs (list (cons bol eol))))
          ;; Blank line: a single segment across the whole row.
          (add bol eol (if (= bol eol) 'before-string 'display) 0 edge)
        ;; Margin, swallowing the line's leading spaces.
        (if (eql (caar runs) bol)
            (let ((run (pop runs)))
              (add bol (cdr run) 'display 0 (funcall column (cdr run))))
          (when (> margin 0)
            (add bol bol 'before-string 0 margin)))
        (let ((tail (car (last runs))))
          (unless (eql (cdr tail) eol)
            (setq tail nil))
          ;; Wide gaps, except the one before a menu key.
          (dolist (run runs)
            (unless (or (eq run tail)
                        (< (- (cdr run) (car run)) my/enlight-rain-min-gap)
                        (eq (get-text-property (cdr run) 'face)
                            'enlight-menu-key))
              (add (car run) (cdr run) 'display (funcall column (car run))
                   (- (cdr run) (car run)))))
          ;; Everything right of the text, up to the window's edge.
          (let ((start (if tail (car tail) eol)))
            (add start eol (if tail 'display 'before-string)
                 (funcall column start)
                 (- edge (funcall column start)))))))
    segments))

(defun my/enlight--rain-layout (size)
  "Rebuild the overlays and the rain for a window of SIZE, given as
(COLUMNS . ROWS)."
  (mapc #'delete-overlay my/enlight--rain-overlays)
  (setq my/enlight--rain-overlays nil
        my/enlight--rain-segments nil
        my/enlight--rain-size size)
  ;; Margins are drawn as rain, and lines never wrap.
  (setq-local line-prefix nil)
  (setq-local truncate-lines t)
  (pcase-let* ((`(,cols . ,rows) size)
               (margin (max 0 (/ (- cols (or enlight-width 0)) 2)))
               (row 0))
    (save-excursion
      (goto-char (point-min))
      (while (< row rows)
        (let ((line (my/enlight--rain-line-segments row margin cols)))
          (setq my/enlight--rain-segments
                (nconc line my/enlight--rain-segments)
                row (1+ row))
          (if (< (line-end-position) (point-max))
              (forward-line 1)
            ;; Out of lines: the last segment also fills the rows below.
            (setf (nth 5 (car line)) (- rows row))
            (setq row rows)))))
    (setq my/enlight--rain-grid
          (vconcat (cl-loop repeat (* cols rows)
                            collect (my/enlight--rain-random-char)))
          my/enlight--rain-drops
          (vconcat (cl-loop repeat cols
                            collect (my/enlight--rain-new-drop rows t))))))

(defun my/enlight--rain-step (seconds)
  "Move every drop on by SECONDS and flicker a few characters."
  (let ((rows (cdr my/enlight--rain-size))
        (grid my/enlight--rain-grid))
    (dotimes (col (length my/enlight--rain-drops))
      (let ((drop (aref my/enlight--rain-drops col)))
        (aset drop 0 (+ (aref drop 0) (* seconds (aref drop 1))))
        (when (> (- (aref drop 0) (aref drop 2)) rows)
          (aset my/enlight--rain-drops col (my/enlight--rain-new-drop rows)))))
    (dotimes (_ (ceiling (* my/enlight-rain-glitch (length grid))))
      (aset grid (random (length grid)) (my/enlight--rain-random-char)))))

(defun my/enlight--rain-row (row col width)
  "WIDTH columns of rain on screen ROW, starting at column COL."
  (let ((text (make-string width ?\s t))
        (cols (car my/enlight--rain-size))
        (levels (length my/enlight--rain-faces)))
    (dotimes (i width)
      (let* ((c (+ col i))
             (drop (aref my/enlight--rain-drops c))
             (behind (- (aref drop 0) row))
             (length (aref drop 2)))
        (when (and (zerop (% c my/enlight-rain-column-step))
                   (>= behind 0) (< behind length))
          (aset text i (aref my/enlight--rain-grid (+ (* row cols) c)))
          (put-text-property
           i (1+ i) 'face
           (if (< behind 1)
               my/enlight--rain-head-face
             (aref my/enlight--rain-faces
                   (min (1- levels)
                        (floor (* levels (- 1 (/ behind length)))))))
           text))))
    text))

(defun my/enlight--rain-paint ()
  "Put the current frame into the overlays."
  (let ((edge (1- (car my/enlight--rain-size))))
    (pcase-dolist (`(,ov ,prop ,row ,col ,width ,below)
                   my/enlight--rain-segments)
      (overlay-put ov prop
                   (apply #'concat
                          (my/enlight--rain-row row col width)
                          (cl-loop for k from 1 to below
                                   collect "\n"
                                   collect (my/enlight--rain-row
                                            (+ row k) 0 edge)))))))

(defun my/enlight--rain-window ()
  "The selected window when it shows the start page, else nil.
While the minibuffer is active, the window it was opened from counts."
  (let ((window (if (minibufferp (window-buffer (selected-window)))
                    (minibuffer-selected-window)
                  (selected-window))))
    (and window
         (buffer-live-p my/enlight--rain-buffer)
         (eq (window-buffer window) my/enlight--rain-buffer)
         window)))

(defun my/enlight--rain-frame ()
  "Advance and draw one frame, or stop when the page is not selected."
  (let ((window (my/enlight--rain-window)))
    (if (not window)
        (my/enlight-rain-stop)
      (with-current-buffer my/enlight--rain-buffer
        (let ((size (cons (window-body-width window)
                          (window-body-height window)))
              (now (current-time)))
          (if (equal size my/enlight--rain-size)
              ;; At most 0.2s of movement per frame.
              (my/enlight--rain-step
               (min 0.2 (float-time (time-subtract now my/enlight--rain-time))))
            (my/enlight--rain-layout size))
          (setq my/enlight--rain-time now)
          (my/enlight--rain-paint))))))

(defun my/enlight-rain-stop ()
  "Pause the rain. The last frame stays on the page."
  (when my/enlight--rain-timer
    (cancel-timer my/enlight--rain-timer)
    (setq my/enlight--rain-timer nil)))

(defun my/enlight--rain-run (delay)
  "Start the frame loop after DELAY seconds, unless it is running."
  (when (and my/enlight-rain-enabled
             (not my/enlight--rain-timer)
             (my/enlight--rain-make-faces))
    (setq my/enlight--rain-timer
          (run-with-timer delay (/ 1.0 my/enlight-rain-fps)
                          #'my/enlight--rain-frame))))

(defun my/enlight-rain ()
  "Start the rain on freshly inserted start page content.
Deletes the overlays from before the redraw."
  (my/enlight-rain-stop)
  (mapc #'delete-overlay my/enlight--rain-overlays)
  (setq my/enlight--rain-overlays nil
        my/enlight--rain-segments nil
        my/enlight--rain-size nil
        my/enlight--rain-buffer (current-buffer))
  (add-hook 'window-buffer-change-functions #'my/enlight--rain-resume nil t)
  (add-hook 'window-selection-change-functions #'my/enlight--rain-resume nil t)
  (my/enlight--rain-run my/enlight-rain-delay))

(defun my/enlight--rain-resume (_window)
  "Restart the rain when the start page becomes the selected window."
  (when (my/enlight--rain-window)
    (my/enlight--rain-run 0)))

(mapc #'byte-compile '(my/enlight--rain-random-char
                       my/enlight--rain-new-drop
                       my/enlight--rain-step
                       my/enlight--rain-row
                       my/enlight--rain-paint
                       my/enlight--rain-window
                       my/enlight--rain-frame))

(add-hook 'enlight-after-insert-hook #'my/enlight-rain)
#+end_src

** Ghostel terminal

Terminal backend for Claude sessions, built on Ghostty's VT engine. Its
native module is downloaded on first use.

=s-p= / =s-n= scroll five steps, and =C-p= / =C-n= go to the terminal.
When the app tracks the mouse, scrolling sends wheel presses (button 4
up, 5 down) at the middle row; otherwise the Emacs window scrolls.

#+begin_src emacs-lisp
(use-package ghostel
  :defer t
  :custom
  (ghostel-module-auto-install 'download)
  :config
  (defun my/ghostel-scroll (button n)
    "Scroll N steps up (BUTTON 4) or down (BUTTON 5).
A step is a wheel press when the app tracks the mouse, else a window line."
    (if (ghostel--mouse-tracking-p ghostel--term)
        (dotimes (_ n)
          (ghostel--mouse-event ghostel--term 0 button
                                (/ (window-body-height) 2) 0 0))
      (if (= button 4) (scroll-down n) (scroll-up n))))
  (defun my/ghostel-scroll-up-5 ()
    (interactive)
    (my/ghostel-scroll 4 5))
  (defun my/ghostel-scroll-down-5 ()
    (interactive)
    (my/ghostel-scroll 5 5))

  (define-key ghostel-mode-map (kbd "s-p") #'my/ghostel-scroll-up-5)
  (define-key ghostel-mode-map (kbd "s-n") #'my/ghostel-scroll-down-5)
  ;; Line motion in copy and Emacs modes.
  (define-key ghostel-readonly-mode-map (kbd "s-p") #'my/previous-line-5)
  (define-key ghostel-readonly-mode-map (kbd "s-n") #'my/next-line-5)
  ;; Sends M-DEL (backward-kill-word).
  (define-key ghostel-mode-map (kbd "s-<backspace>")
              (lambda () (interactive) (ghostel-send-key "backspace" "meta"))))

** claude-code-ide

Claude Code sessions inside Emacs, opened from =C-c C-'= or =M-x claude=,
with the Emacs MCP tools enabled. =C-c c= skips the menu and starts a
new instance, and =C-c n= starts one in a new frame.

#+begin_src emacs-lisp
(use-package claude-code-ide
  :vc (:url "https://github.com/manzaltu/claude-code-ide.el" :rev :newest)
  :bind (("C-c C-'" . claude-code-ide-menu)
         ("C-c c" . claude-code-ide)
         ("C-c n" . my/claude-new-frame))
  :custom
  (claude-code-ide-terminal-backend 'ghostel)
  ;; Diffs open in their own frame (see below).
  (claude-code-ide-show-claude-window-in-ediff nil)
  :init
  (defalias 'claude #'claude-code-ide-menu)
  :config
  ;; Defines the `json-rpc-error' signalled by the MCP HTTP server.
  (define-error 'json-rpc-error "JSON-RPC error" 'error)

  ;; Answer the resources and prompts list requests with empty collections
  ;; instead of "Method not found".
  (advice-add 'claude-code-ide-mcp-http-server--dispatch :around
              (lambda (orig method params)
                (pcase method
                  ("resources/list" '((resources . [])))
                  ("resources/templates/list" '((resourceTemplates . [])))
                  ("prompts/list" '((prompts . [])))
                  (_ (funcall orig method params)))))

  ;; Diff frame filling the left half of the monitor's work area, outer
  ;; size included. Remembers the frame it was opened from.
  (defun my/claude-diff-make-frame ()
    (pcase-let* ((`(,x ,y ,width ,height) (frame-monitor-workarea))
                 (frame (make-frame `((name . "Claude diff")
                                      (fullscreen . nil)
                                      (my/claude-diff-origin . ,(selected-frame))))))
      (set-frame-position frame x y)
      (set-frame-size frame
                      (- (/ width 2) (- (frame-outer-width frame) (frame-text-width frame)))
                      (- height (- (frame-outer-height frame) (frame-text-height frame)))
                      t)
      frame))

  ;; Opens each diff in a new frame, tagged with its session and tab name.
  ;; A stale file buffer without unsaved edits is reverted first. The
  ;; frame is deleted if the handler exits abnormally, including C-g.
  (defun my/claude-diff-in-new-frame (orig arguments &optional session)
    (let ((buffer (find-buffer-visiting (alist-get 'old_file_path arguments)))
          (opened nil)
          frame)
      (when (and buffer
                 (not (buffer-modified-p buffer))
                 (not (verify-visited-file-modtime buffer)))
        (with-current-buffer buffer
          (revert-buffer t t t)))
      (setq frame (my/claude-diff-make-frame))
      (select-frame-set-input-focus frame)
      (unwind-protect
          (prog1 (let ((claude-code-ide-switch-tab-on-ediff nil))
                   (funcall orig arguments session))
            (setq opened t)
            (set-frame-parameter frame 'my/claude-diff
                                 (cons session (alist-get 'tab_name arguments))))
        (unless opened
          (my/claude-diff-close-frame frame)))))

  ;; Deletes FRAME and gives input focus back to its origin frame.
  (defun my/claude-diff-close-frame (frame)
    (let ((origin (frame-parameter frame 'my/claude-diff-origin)))
      (delete-frame frame)
      (when (frame-live-p origin)
        (select-frame-set-input-focus origin))))

  (defun my/claude-diff-frame (tab-name session)
    (seq-find (lambda (frame)
                (let ((tag (frame-parameter frame 'my/claude-diff)))
                  (and (eq (car tag) session) (equal (cdr tag) tab-name))))
              (frame-list)))

  ;; Runs close_tab with the diff frame selected.
  (defun my/claude-diff-close-in-frame (orig arguments &optional session)
    (if-let* ((frame (my/claude-diff-frame (alist-get 'tab_name arguments) session)))
        (with-selected-frame frame
          (funcall orig arguments session))
      (funcall orig arguments session)))

  ;; Deletes the diff frame when cleanup-diff runs.
  (defun my/claude-diff-delete-frame (tab-name session)
    (when-let* ((frame (my/claude-diff-frame tab-name session)))
      (my/claude-diff-close-frame frame)))

  (advice-add 'claude-code-ide-mcp-handle-open-diff :around #'my/claude-diff-in-new-frame)
  (advice-add 'claude-code-ide-mcp-handle-close-tab :around #'my/claude-diff-close-in-frame)
  (advice-add 'claude-code-ide-mcp--cleanup-diff :after #'my/claude-diff-delete-frame)

  ;; Starts a new instance filling a frame of its own, using the calling
  ;; buffer's project. The frame is deleted if the start fails or is quit,
  ;; and when the instance's buffer is killed.
  (defun my/claude-new-frame ()
    (interactive)
    (let ((origin (current-buffer))
          (frame (make-frame '((name . "Claude"))))
          (started nil))
      (select-frame-set-input-focus frame)
      (unwind-protect
          (let ((claude-code-ide-use-side-window nil)
                (display-buffer-overriding-action '(display-buffer-same-window)))
            (with-current-buffer origin
              (claude-code-ide))
            (setq started t)
            (with-current-buffer (window-buffer (frame-selected-window frame))
              (add-hook 'kill-buffer-hook
                        (lambda ()
                          (when (and (frame-live-p frame) (cdr (frame-list)))
                            (delete-frame frame)))
                        nil t)))
        (unless started
          (delete-frame frame)))))

  ;; Starts the Emacs server before every session launch, for emacsclient.
  ;; Advises the private `claude-code-ide--start-session'.
  (defun my/claude-ensure-server (&rest _)
    (require 'server)
    (unless (server-running-p)
      (server-start)))

  (advice-add 'claude-code-ide--start-session :before #'my/claude-ensure-server)

  (claude-code-ide-emacs-tools-setup))
